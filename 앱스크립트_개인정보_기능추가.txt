var PRIVACY_SHEET = "개인정보"; // 학번/비밀번호가 있는 시트 이름

/**
 * 로그인 검증 (학번, 비번 일치 여부 확인)
 * E열(5): 학번, F열(6): 초기비번, G열(7): 사용자수정비번
 */
function verifyUserAuth(userId, userPwd) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(PRIVACY_SHEET);
  if (!sheet) return { success: false, message: "개인정보 시트가 존재하지 않습니다." };

  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) { // 헤더(1행) 제외
    var sheetId = String(data[i][4]).trim(); // E열 (0-indexed로 4)
    if (sheetId === String(userId).trim()) {
      var initialPwd = String(data[i][5]).trim(); // F열 (0-indexed로 5)
      var changedPwd = String(data[i][6]).trim(); // G열 (0-indexed로 6)
      var activePwd = changedPwd ? changedPwd : initialPwd; // G열 값이 있으면 G열 우선, 없으면 F열

      if (activePwd === String(userPwd).trim()) {
        return { success: true };
      } else {
        return { success: false, message: "비밀번호가 일치하지 않습니다." };
      }
    }
  }
  return { success: false, message: "존재하지 않는 아이디(학번)입니다." };
}

/**
 * 비밀번호 변경
 */
function changeUserPassword(userId, oldPwd, newPwd) {
  var authCheck = verifyUserAuth(userId, oldPwd);
  if (!authCheck.success) {
    return { success: false, message: "기존 비밀번호가 일치하지 않거나 아이디가 존재하지 않습니다." };
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(PRIVACY_SHEET);
  var data = sheet.getDataRange().getValues();
  
  for (var i = 1; i < data.length; i++) {
    var sheetId = String(data[i][4]).trim(); // E열
    if (sheetId === String(userId).trim()) {
      // 행 번호는 배열 인덱스(i) + 1, G열은 7번째 열
      sheet.getRange(i + 1, 7).setValue(newPwd); 
      return { success: true };
    }
  }
  return { success: false, message: "비밀번호 변경 중 알 수 없는 오류 발생" };
}
