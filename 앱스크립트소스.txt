// [환경 설정] 전역 변수 (Global Variables)
// 다른 파일들과 충돌하지 않도록 고유한 이름을 사용합니다.
// ---------------------------------------------------------
var MATH_BATTLE_V1_TOKEN = "super_secret_key_2026!"; // 완전제곱식 게임 보안 키
var MATH_BATTLE_V1_SHEET = "완전제곱식 게임1";         // 완전제곱식 데이터를 저장할 시트 이름
var MATH_BATTLE_V2_TOKEN = "super_secret_key_20262"; // 일차방정식 게임 보안 키
var SYSTEM_CONFIG_SHEET = "시스템 설정";             // 입장코드를 분리 보관할 전용 시트

// ---------------------------------------------------------
// [1] 통합 doPost 함수 (게임 결과 데이터 저장/삭제 전용)
// ---------------------------------------------------------
function doPost(e) {
  var payload; 
  try {
    payload = JSON.parse(e.postData.contents);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, message: "JSON Parse Error" }))
        .setMimeType(ContentService.MimeType.JSON);
  }

  // --- 1. 일차방정식 게임의 요청인 경우 (라우팅) ---
  if (payload.gameId === "일차방정식 게임1") {
    return processLinearGamePost(payload);
  }

  // --- 2. 기존 완전제곱식 게임 로직 처리 ---
  try {
    if (payload.token !== MATH_BATTLE_V1_TOKEN) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Invalid token" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (payload.action === 'save') {
      var doc = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = doc.getSheetByName(MATH_BATTLE_V1_SHEET);
      
      if (!sheet) {
        sheet = doc.insertSheet(MATH_BATTLE_V1_SHEET);
        sheet.appendRow(["시간", "용사이름", "레벨", "처치 몬스터", "경험치", "획득한 동료 이미지"]);
        sheet.getRange("A1:F1").setFontWeight("bold").setBackground("#d9ead3");
      }

      var kstDate = Utilities.formatDate(new Date(payload.timestamp), "Asia/Seoul", "yyyy.MM.dd HH:mm:ss");

      sheet.appendRow([
        kstDate,          
        payload.name,     
        payload.level,    
        payload.score,    
        payload.xp,       
        payload.captured  
      ]);

      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (payload.action === 'delete') {
      var doc = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = doc.getSheetByName(MATH_BATTLE_V1_SHEET);
      if (sheet) {
        var kstDateToDelete = Utilities.formatDate(new Date(payload.timestamp), "Asia/Seoul", "yyyy.MM.dd HH:mm:ss");
        var values = sheet.getDataRange().getValues();
        
        for (var i = values.length - 1; i >= 1; i--) {
          var rowTime = values[i][0]; 
          if (rowTime instanceof Date) {
            rowTime = Utilities.formatDate(rowTime, "Asia/Seoul", "yyyy.MM.dd HH:mm:ss");
          }
          if (rowTime === kstDateToDelete || rowTime == payload.timestamp) {
            sheet.deleteRow(i + 1); 
            break; 
          }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ---------------------------------------------------------
// [2] 일차방정식 게임 전용 POST 처리 함수
// ---------------------------------------------------------
function processLinearGamePost(payload) {
  try {
    var action = payload.action;
    var gameId = payload.gameId; 
    var token = payload.token;

    if (token !== MATH_BATTLE_V2_TOKEN) {
      return ContentService.createTextOutput(JSON.stringify({ error: "Invalid token" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(gameId);
    
    if (!sheet) {
      sheet = ss.insertSheet(gameId);
      sheet.appendRow(["시간", "용사이름", "처치 몬스터", "레벨", "경험치", "획득한 동료 이미지"]);
      sheet.getRange("A1:F1").setFontWeight("bold").setBackground("#cfe2f3"); 
    }

    if (action === "save") {
      var kstDate = Utilities.formatDate(new Date(payload.timestamp), "Asia/Seoul", "yyyy.MM.dd HH:mm:ss");
      sheet.appendRow([
        kstDate, 
        payload.name, 
        payload.score, 
        payload.level, 
        payload.xp, 
        payload.captured
      ]);
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);

    } else if (action === "delete") {
      var data = sheet.getDataRange().getValues();
      var kstDateToDelete = Utilities.formatDate(new Date(payload.timestamp), "Asia/Seoul", "yyyy.MM.dd HH:mm:ss");
      
      for (var i = data.length - 1; i >= 1; i--) {
        var rowTime = data[i][0];
        if (rowTime instanceof Date) {
          rowTime = Utilities.formatDate(rowTime, "Asia/Seoul", "yyyy.MM.dd HH:mm:ss");
        }
        if (rowTime === kstDateToDelete || rowTime == payload.timestamp) {
          sheet.deleteRow(i + 1); 
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ---------------------------------------------------------
// [3] 통합 doGet 함수 (데이터 로드 & 입장코드 획득 및 저장 GET 대응)
// ---------------------------------------------------------
function doGet(e) {
  
  // [치명적 버그 수정 완료] 파라미터 값을 함수 최상단에서 먼저 할당!
  var action = e.parameter.action;
  var gameId = e.parameter.gameId; // [추가] 반복되는 gameId도 미리 변수로 빼둡니다.

  // --- [수정 완료] 입장 코드 저장 (시스템 설정 시트의 해당 게임 열에 누적 저장) ---
  // [신규 추가] 설정값 시트 저장 액션
  if (action === 'saveSettings') {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = ss.getSheetByName("시스템 설정");
      
      // '시스템 설정' 시트가 없다면 자동으로 생성합니다.
      if (!sheet) {
        sheet = ss.insertSheet("시스템 설정");
        sheet.getRange("D1").setValue("=== 설정 히스토리 ==="); // D1에 헤더 작성
      }
      
      var deployId = e.parameter.saveDeployId;
      var gameId = e.parameter.saveGameId;
      
      // D열에서 데이터가 있는 마지막 행(row) 번호를 찾습니다.
      var dColumnValues = sheet.getRange("D:D").getValues();
      var lastRow = 0;
      for (var i = dColumnValues.length - 1; i >= 0; i--) {
        if (dColumnValues[i][0] !== "") {
          lastRow = i + 1;
          break;
        }
      }
      
      // 질문자님 요청: D2, D3 -> 다음엔 D4, D5 순으로 차곡차곡 쌓이게 작성
      var nextRowForDeployId = lastRow + 1;
      if (nextRowForDeployId === 1) nextRowForDeployId = 2; // 안전장치: 1행은 헤더
      
      var nextRowForGameId = nextRowForDeployId + 1;
      
      // 시트에 값 기록
      sheet.getRange("D" + nextRowForDeployId).setValue(deployId);
      sheet.getRange("D" + nextRowForGameId).setValue(gameId);
      
      return ContentService.createTextOutput(JSON.stringify({success: true}))
                           .setMimeType(ContentService.MimeType.JSON);
    } catch(err) {
      return ContentService.createTextOutput(JSON.stringify({success: false, error: err.toString()}))
                           .setMimeType(ContentService.MimeType.JSON);
    }
  }

  if (action === 'setAccessCode') {
    var expectedToken = (gameId === "일차방정식 게임1") ? MATH_BATTLE_V2_TOKEN : MATH_BATTLE_V1_TOKEN;
    
    if (e.parameter.token !== expectedToken) {
      return ContentService.createTextOutput(JSON.stringify({success: false, error: "Invalid Token"}))
          .setMimeType(ContentService.MimeType.JSON);
    }
    
    try {
      saveAccessCodeToConfig(gameId, e.parameter.accessCode);
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
          .setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
          .setMimeType(ContentService.MimeType.JSON);
    }
  }

  // --- [수정 완료] 기존 입장 코드 불러오기 (해당 게임 열의 가장 밑에 있는 값 가져오기) ---
  if (action === 'getAccessCode') {
    try {
      var code = getAccessCodeFromConfig(gameId);
      return ContentService.createTextOutput(JSON.stringify({
        success: true, 
        accessCode: code
      })).setMimeType(ContentService.MimeType.JSON);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
          .setMimeType(ContentService.MimeType.JSON);
    }
  }

  // --- 1. 일차방정식 게임 라우팅 ---
  if (gameId === "일차방정식 게임1") {
    return processLinearGameGet(e.parameter);
  }
  
  // --- 2. 기존 완전제곱식 게임 결과 로드 ---
  try {
        
    if (action === 'load') {
      var doc = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = doc.getSheetByName(MATH_BATTLE_V1_SHEET);
      
      if (!sheet) {
        return ContentService.createTextOutput(JSON.stringify([]))
          .setMimeType(ContentService.MimeType.JSON);
      }
      
      var dataRange = sheet.getDataRange();
      var values = dataRange.getValues();
      var records = [];
      
      for (var i = 1; i < values.length; i++) {
        var row = values[i];
        if (!row[1]) continue; 
        
        records.push({
          timestamp: row[0], 
          name: row[1],      
          level: row[2],     
          score: row[3],     
          xp: row[4],        
          captured: row[5]   
        });
      }
      
      records.sort(function(a, b) { return b.xp - a.xp; });

      return ContentService.createTextOutput(JSON.stringify(records))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput("서버가 정상적으로 동작 중입니다.");
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// [4] 일차방정식 게임 전용 GET 처리 함수
// ---------------------------------------------------------
function processLinearGameGet(parameter) {
  try {
    var action = parameter.action;
    var gameId = parameter.gameId;

    if (action === "load") {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = ss.getSheetByName(gameId);
      
      if (!sheet) {
        return ContentService.createTextOutput(JSON.stringify([]))
          .setMimeType(ContentService.MimeType.JSON);
      }

      var data = sheet.getDataRange().getValues();
      var records = [];
      
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        if (!row[1]) continue;
        
        records.push({
          timestamp: row[0],
          name: row[1],
          score: row[2], 
          level: row[3], 
          xp: row[4],
          captured: row[5]
        });
      }
      
      records.sort(function(a, b) { return b.xp - a.xp; });

      return ContentService.createTextOutput(JSON.stringify(records))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ---------------------------------------------------------
// [내부 헬퍼 함수] 시스템 설정 시트에 입장코드를 누적 기록하는 함수
// ---------------------------------------------------------
function saveAccessCodeToConfig(gameId, accessCode) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SYSTEM_CONFIG_SHEET);
  if (!sheet) {
    sheet = ss.insertSheet(SYSTEM_CONFIG_SHEET);
  }
  
  var lastCol = sheet.getLastColumn();
  var colIndex = -1;
  
  if (lastCol > 0) {
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    for (var j = 0; j < headers.length; j++) {
      if (headers[j] === gameId) {
        colIndex = j + 1;
        break;
      }
    }
  }
  
  // 열이 없으면 1행에 게임 ID(예: 완전제곱식 게임1) 헤더를 적고 새 열 생성
  if (colIndex === -1) {
    colIndex = lastCol + 1;
    sheet.getRange(1, colIndex).setValue(gameId);
    sheet.getRange(1, colIndex).setFontWeight("bold").setBackground("#fce5cd");
  }
  
  // 해당 열의 데이터를 가져와 역순으로 검색하여 데이터가 존재하는 가장 아래 행을 탐색합니다.
  var lastRow = sheet.getLastRow();
  var nextRow = 2; // 기본적으로 2행부터 쓰기 시작
  if (lastRow > 0) {
    var colValues = sheet.getRange(1, colIndex, lastRow, 1).getValues();
    for (var i = colValues.length - 1; i >= 0; i--) {
      if (colValues[i][0] !== "") {
        nextRow = i + 2;
        break;
      }
    }
  }
  
  sheet.getRange(nextRow, colIndex).setValue(accessCode);
}

// ---------------------------------------------------------
// [내부 헬퍼 함수] 시스템 설정 시트에서 해당 게임의 가장 아래 코드를 가져오는 함수
// ---------------------------------------------------------
function getAccessCodeFromConfig(gameId) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SYSTEM_CONFIG_SHEET);
  if (!sheet) return "";
  
  var lastCol = sheet.getLastColumn();
  if (lastCol === 0) return "";
  
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var colIndex = -1;
  for (var j = 0; j < headers.length; j++) {
    if (headers[j] === gameId) {
      colIndex = j + 1;
      break;
    }
  }
  
  if (colIndex === -1) return "";
  
  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) return ""; // 헤더 외에 데이터가 없음
  
  var colValues = sheet.getRange(1, colIndex, lastRow, 1).getValues();
  // 밑에서부터 역순으로 탐색하여 최초로 공백이 아닌 입장코드를 읽어옵니다.
  for (var i = colValues.length - 1; i >= 1; i--) {
    if (colValues[i][0] !== "") {
      return String(colValues[i][0]).trim();
    }
  }
  return "";
}




/**
 * 클라이언트(리액트 등)에서 POST 요청을 보낼 때 실행되는 함수입니다.
 * 주로 데이터를 기록하거나 삭제할 때 사용됩니다.
 */
function doPost(e1) {
  try {
    // 1. 활성화된 스프레드시트에서 "좌표1"이라는 이름의 시트를 가져옵니다.
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("좌표1");
    
    // 2. 만약 "좌표1" 시트가 존재하지 않는다면 새로 만들고 헤더(첫 줄)를 작성합니다.
    if (!sheet) {
      sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("좌표1");
      sheet.appendRow(["등록 시간", "이름", "점수", "정답률", "걸린 시간"]);
    }
    
    // 3. 클라이언트가 보낸 데이터(JSON 형식)를 자바스크립트 객체로 변환합니다.
    var params = JSON.parse(e1.postData.contents);
    
    // 어떤 작업(action)을 할 것인지 변수에 담습니다. ("register" 또는 "delete")
    var action = params.action;
    
    // =================================================================
    // [액션 1] 데이터 등록 (명예의 전당에 새로운 기록 추가)
    // =================================================================
    if (action === "register") {
      // 한국 시간(KST) 기준으로 현재 시간을 "YYYY-MM-DD HH:mm:ss" 형태로 만듭니다.
      var kstDate = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss");
      
      // 전달받은 값이 없으면 기본값을 사용하도록 안전하게 처리합니다.
      var name = params.name || "무명 훈련사";
      var score = Number(params.score) || 0;
      var accuracy = params.accuracy !== undefined ? Number(params.accuracy) : 100;
      var time = Number(params.time) || 0; 
      
      // 시트의 맨 아래에 새로운 행(데이터)을 추가합니다.
      sheet.appendRow([kstDate, name, score, accuracy, time]);
      
      // 성공했다는 메시지를 클라이언트(리액트)로 다시 보내줍니다.
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        message: "명예의 전당에 성공적으로 등록되었습니다!"
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // =================================================================
    // [액션 2] 데이터 삭제 (관리자가 휴지통 버튼을 눌렀을 때)
    // =================================================================
    else if (action === "delete") {
      // 삭제하고자 하는 타겟 데이터를 변수에 담습니다.
      var nameToDelete = params.name;
      var scoreToDelete = Number(params.score);
      var timeToDelete = Number(params.time);
      
      // 시트의 모든 데이터를 2차원 배열 형태로 가져옵니다.
      var data = sheet.getDataRange().getValues();
      var deletedCount = 0;
      
      // 💡 아주 중요한 로직: 행을 삭제할 때는 반드시 맨 아래에서부터 위로 올라가며 검사해야 합니다!
      // 위에서부터 지우면 행 번호가 하나씩 위로 당겨져서(Index Shift) 엉뚱한 줄을 지우거나 건너뛰게 됩니다.
      // i >= 1인 이유는 i가 0일 때(헤더행: 이름, 점수 등)는 지우지 않기 위함입니다.
      for (var i = data.length - 1; i >= 1; i--) {
        // data[i]는 특정 행 전체를 의미하며, 인덱스 1은 두 번째 열(이름)입니다.
        var rowName = data[i][1];
        var rowScore = Number(data[i][2]);
        var rowTime = Number(data[i][4]);
        
        // 전달받은 타겟의 이름, 점수, 걸린 시간이 시트의 줄과 모두 일치하는지 확인합니다.
        if (rowName === nameToDelete && rowScore === scoreToDelete && rowTime === timeToDelete) {
          
          // 앱스크립트에서 시트의 행 번호는 1부터 시작하고, 배열(data)의 인덱스는 0부터 시작합니다.
          // 따라서 i번 인덱스의 데이터를 지우려면 시트에서는 i + 1번 행을 지워야 합니다.
          sheet.deleteRow(i + 1); 
          deletedCount++;
          
          // 동명이인이 같은 점수/시간을 가질 확률은 적지만, 중복 삭제를 막기 위해
          // 일치하는 데이터를 1개만 지우고 반복문을 즉시 종료(break)합니다.
          break; 
        }
      }
      
      // 삭제가 완료되면 클라이언트에게 결과를 알려줍니다.
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        message: deletedCount + "개의 기록이 성공적으로 삭제되었습니다."
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
  } catch (error) {
    // 에러가 발생하면 에러 내용을 클라이언트로 보냅니다.
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}


/**
 * 클라이언트(리액트 등)에서 GET 요청을 보낼 때 실행되는 함수입니다.
 * 주로 명예의 전당 랭킹 데이터를 읽어갈 때 사용됩니다.
 */
function doGet(e1) {
  try {
    // 1. "좌표1" 시트를 가져옵니다.
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("좌표1");
    
    // 시트가 존재하면 데이터 전체를 가져오고, 없으면 빈 배열을 준비합니다.
    var data = sheet ? sheet.getDataRange().getValues() : [];
    var rankings = [];
    
    // 2. 데이터를 순회하며 랭킹 배열을 만듭니다. (i = 1부터 시작하여 첫 줄 헤더는 건너뜁니다)
    for (var i = 1; i < data.length; i++) {
      rankings.push({
        date: data[i][0],              // A열: 등록 시간
        name: data[i][1],              // B열: 이름
        score: Number(data[i][2]),     // C열: 점수 (숫자로 변환)
        accuracy: Number(data[i][3]),  // D열: 정답률 (숫자로 변환)
        time: Number(data[i][4])       // E열: 걸린 시간 (숫자로 변환)
      });
    }
    
    // 3. 랭킹 규칙에 맞춰 데이터를 정렬(Sort)합니다.
    rankings.sort(function(a, b) {
      // 만약 점수가 같다면?
      if (b.score === a.score) {
        // 걸린 시간이 더 짧은(숫자가 작은) 사람이 상위에 오르도록 오름차순 정렬
        return a.time - b.time; 
      }
      // 기본 규칙: 점수가 높은 사람이 상위에 오르도록 내림차순 정렬
      return b.score - a.score;
    });
    
    // 4. 정렬된 전체 데이터 중 1등부터 10등까지만 잘라냅니다(slice).
    var top10 = rankings.slice(0, 10);
    
    // 5. 최종 완성된 10명의 데이터를 클라이언트로 보내줍니다.
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      data: top10
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    // 에러 발생 시 처리
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * 💡 몬스터 좌표 찾기 Pro - 통합 백엔드 코어
 * 대상 시트명: "좌표2", "시스템 설정"
 */

var SECRET_TOKEN = "super_secret_key_2026!"; // 프론트엔드 보안 키와 일치해야 합니다.
var TARGET_SHEET = "좌표2";
var SYSTEM_SHEET = "시스템 설정";

// ---------------------------------------------------------
// [1] POST 요청 처리 (게임 기록 등록 및 관리자 삭제)
// ---------------------------------------------------------
function doPost(e2) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TARGET_SHEET);
    if (!sheet) {
      sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet(TARGET_SHEET);
      sheet.appendRow(["등록 시간", "이름", "점수", "정답률", "걸린 시간"]);
    }
    
    var params = JSON.parse(e2.postData.contents);
    var action = params.action;
    
    // [액션 1] 게임 기록 등록
    if (action === "register") {
      var kstDate = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss");
      var name = params.name || "무명 용사";
      var score = Number(params.score) || 0;
      var accuracy = params.accuracy !== undefined ? Number(params.accuracy) : 100;
      var time = Number(params.time) || 0; 
      
      sheet.appendRow([kstDate, name, score, accuracy, time]);
      
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        message: "명예의 전당에 성공적으로 등록되었습니다!"
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // [액션 2] 명예의 전당 기록 삭제
    else if (action === "delete") {
      var nameToDelete = params.name;
      var scoreToDelete = Number(params.score);
      var timeToDelete = Number(params.time);
      
      var data = sheet.getDataRange().getValues();
      var deletedCount = 0;
      
      for (var i = data.length - 1; i >= 1; i--) {
        if (data[i][1] === nameToDelete && Number(data[i][2]) === scoreToDelete && Number(data[i][4]) === timeToDelete) {
          sheet.deleteRow(i + 1);
          deletedCount++;
          break; // 중복 삭제 방지
        }
      }
      
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        message: deletedCount + "개의 기록이 삭제되었습니다."
      })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

// ---------------------------------------------------------
// [2] GET 요청 처리 (데이터 동기화, 수정 및 불러오기)
// ---------------------------------------------------------
function doGet(e2) {
  try {
    var action = e2.parameter.action;
    
    // [액션 1] 명예의 전당 랭킹 불러오기
    if (action === "getRankings" || action === "load") {
      var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TARGET_SHEET);
      var data = sheet ? sheet.getDataRange().getValues() : [];
      var rankings = [];
      
      for (var i = 1; i < data.length; i++) {
        rankings.push({
          timestamp: data[i][0] ? new Date(data[i][0]).getTime() : Date.now(),
          name: data[i][1],
          score: Number(data[i][2]),
          accuracy: Number(data[i][3]),
          time: Number(data[i][4])
        });
      }
      
      rankings.sort(function(a, b) {
        if (b.score === a.score) return a.time - b.time; 
        return b.score - a.score;
      });
      
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        success: true,
        data: rankings.slice(0, 10)
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // [액션 2] 현재 입장 코드 조회
    else if (action === "getAccessCode") {
      var gameId = e2.parameter.gameId || "default";
      var code = PropertiesService.getScriptProperties().getProperty("access_code_" + gameId) || "";
      
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        accessCode: code
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // [액션 3] 입장 코드 설정 (수정)
    else if (action === "setAccessCode") {
      var gameId = e2.parameter.gameId || "default";
      var newCode = e2.parameter.accessCode;
      var token = e2.parameter.token;
      
      if (token !== SECRET_TOKEN) {
        return ContentService.createTextOutput(JSON.stringify({
          success: false,
          error: "인증 보안 토큰 불일치"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      
      PropertiesService.getScriptProperties().setProperty("access_code_" + gameId, newCode);
      
      return ContentService.createTextOutput(JSON.stringify({
        success: true
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // [액션 4] 시스템 설정 값 시트에 백업
    else if (action === "saveSettings") {
      var deployId = e2.parameter.saveDeployId;
      var saveGameId = e2.parameter.saveGameId;
      
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sysSheet = ss.getSheetByName(SYSTEM_SHEET);
      if (!sysSheet) {
        sysSheet = ss.insertSheet(SYSTEM_SHEET);
        sysSheet.getRange("A1:B1").setValues([["배포 ID", "게임 ID"]]);
      }
      sysSheet.appendRow([deployId, saveGameId]);
      
      return ContentService.createTextOutput(JSON.stringify({
        success: true
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // 매칭되는 액션이 없을 때
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: "요청한 액션을 찾을 수 없습니다."
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: error.toString(),
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
